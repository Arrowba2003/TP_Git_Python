# 1. Initialisation du projet

# pwd : afficher le chemin du dossier actuel
# git init : initialiser un nouveau depot Git
# git status : verifier l'etat des fichiers

# 2. Configuration de Git

# git config --global user.name : configurer le nom de l'utilisateur
# git config --global user.email : configurer l'adresse email

# 3. Creation des versions

# git add main.py : ajouter le fichier a la preparation
# git commit -m : enregistrer les modifications
# git log : afficher l'historique des versions

# 4. Connexion avec GitHub

# git branch -M main : renommer la branche en main
# git remote add origin URL : connecter le projet a GitHub
# git remote -v : afficher l'adresse du depot distant
# git push -u origin main : envoyer le projet vers GitHub

# 5. Synchronisation du projet

# git pull origin main : recuperer les modifications de GitHub
# git push : envoyer les nouvelles modifications vers GitHub

# 6. Execution du programme

# py main.py : executer le programme Python

# 7. Questions de comprehension

# git push : envoie les modifications du PC vers GitHub
# git pull : recupere les modifications de GitHub vers le PC
# origin : nom du depot distant
# git add : prepare les fichiers avant le commit
# git commit : enregistre une version du projet

# Si les deux depots sont deja synchronises,
# git pull indique que tout est deja a jour.
